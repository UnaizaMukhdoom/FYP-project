rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Users collection
    match /users/{userId} {
      // Allow users to read/write their own document
      allow read, write: if request.auth != null && request.auth.uid == userId;
      
      // Allow authenticated users to read all user documents (for admin panel)
      allow read: if request.auth != null;
      
      // Subcollections under users
      match /{document=**} {
        // Allow users to read/write their own subcollections
        allow read, write: if request.auth != null && request.auth.uid == userId;
        
        // Allow authenticated users to read all subcollections (for admin panel)
        allow read: if request.auth != null;
      }
    }
    
    // Config collection (if needed for color analysis)
    match /config/{document=**} {
      allow read: if request.auth != null;
    }
    
    // Questionnaire collection
    match /questionnaire_v1/{document=**} {
      allow read: if request.auth != null;
    }
    
    // Chatbot interactions (if you have this collection)
    match /chatbot_interactions/{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
